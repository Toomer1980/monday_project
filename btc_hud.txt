import tkinter as tk
import requests
import threading
import time

def get_btc_price():
    try:
        response = requests.get("https://api.coindesk.com/v1/bpi/currentprice.json")
        price = response.json()['bpi']['USD']['rate_float']
        return f"BTC: ${price:,.2f}"
    except: return "BTC: OFFLINE"

def update():
    while True:
        label.config(text=get_btc_price())
        time.sleep(30)

root = tk.Tk()
root.overrideredirect(True)
root.attributes("-topmost", True)
root.attributes("-transparentcolor", "black")
root.config(bg="black")
root.geometry(f"300x50+{root.winfo_screenwidth()-320}+50")
label = tk.Label(root, text="SCANNING...", font=("Consolas", 20, "bold"), fg="#00FF00", bg="black")
label.pack()
threading.Thread(target=update, daemon=True).start()
root.mainloop()
